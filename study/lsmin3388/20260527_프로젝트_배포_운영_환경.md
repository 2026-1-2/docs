## 2. Docker Compose 운영 설정

### 2-1. 운영용 docker-compose.prod.yml

```yaml
version: '3.8'

services:
  mmp-server:
    image: mmp-server:latest
    restart: always
    deploy:
      resources:
        limits:
          cpus: '2.0'
          memory: 1G
        reservations:
          cpus: '0.5'
          memory: 256M
    logging:
      driver: json-file
      options:
        max-size: "50m"
        max-file: "5"
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:3000/api/health"]
      interval: 30s
      timeout: 10s
      retries: 3
      start_period: 40s
    env_file:
      - .env.production
    ports:
      - "3000:3000"

  mmp-client:
    image: mmp-client:latest
    restart: always
    deploy:
      resources:
        limits:
          cpus: '0.5'
          memory: 256M
    logging:
      driver: json-file
      options:
        max-size: "20m"
        max-file: "3"
    ports:
      - "80:80"
      - "443:443"
    volumes:
      - ./nginx/conf.d:/etc/nginx/conf.d:ro
      - ./certbot/www:/var/www/certbot:ro
      - ./certbot/conf:/etc/letsencrypt:ro

  mediamtx:
    image: bluenviron/mediamtx:latest
    restart: unless-stopped
    deploy:
      resources:
        limits:
          cpus: '4.0'
          memory: 2G
    logging:
      driver: json-file
      options:
        max-size: "100m"
        max-file: "5"
    ports:
      - "8554:8554"   # RTSP
      - "8888:8888"   # HLS
      - "8889:8889"   # WebRTC
    volumes:
      - ./mediamtx.yml:/mediamtx.yml:ro

  mysql:
    image: mysql:8.0
    restart: always
    deploy:
      resources:
        limits:
          cpus: '2.0'
          memory: 2G
    logging:
      driver: json-file
      options:
        max-size: "50m"
        max-file: "5"
    volumes:
      - mysql-data:/var/lib/mysql
      - ./mysql/conf.d:/etc/mysql/conf.d:ro
    env_file:
      - .env.production
    healthcheck:
      test: ["CMD", "mysqladmin", "ping", "-h", "localhost"]
      interval: 30s
      timeout: 10s
      retries: 5

  video-recorder:
    image: video-recorder:latest
    restart: always
    deploy:
      resources:
        limits:
          cpus: '4.0'
          memory: 2G
    volumes:
      - recordings:/recordings
    env_file:
      - .env.production

volumes:
  mysql-data:
  recordings:
```

### 2-2. Restart Policy 비교

| 정책 | 동작 | 적용 대상 |
|------|------|----------|
| `no` | 재시작하지 않음 | 일회성 작업 |
| `always` | 항상 재시작 | DB, API 서버 |
| `unless-stopped` | 수동 중지 전까지 재시작 | MediaMTX |
| `on-failure` | 오류 종료 시만 재시작 | 배치 작업 |

### 4-3. 헬스체크 API

```typescript
// mmp-server: health.controller.ts
@Controller('api/health')
export class HealthController {
  constructor(
    private readonly prisma: PrismaService,
    private readonly mediaService: MediaService,
  ) {}

  @Get()
  async check() {
    const dbOk = await this.checkDatabase();
    const mediamtxOk = await this.checkMediaMTX();
    const diskOk = await this.checkDisk();

    const status = dbOk && mediamtxOk && diskOk ? 'healthy' : 'unhealthy';
    return {
      status,
      timestamp: new Date().toISOString(),
      checks: {
        database: dbOk ? 'up' : 'down',
        mediamtx: mediamtxOk ? 'up' : 'down',
        disk: diskOk ? 'ok' : 'critical',
      },
    };
  }

  private async checkDatabase(): Promise<boolean> {
    try {
      await this.prisma.$queryRaw`SELECT 1`;
      return true;
    } catch {
      return false;
    }
  }

  private async checkDisk(): Promise<boolean> {
    const { execSync } = require('child_process');
    const usage = parseInt(
      execSync("df /recordings | tail -1 | awk '{print $5}'").toString().replace('%', '')
    );
    return usage < 90;
  }
}
```

## 6. 백업 전략

### 6-1. MySQL 데이터 백업

```bash
#!/bin/bash
# backup-mysql.sh

BACKUP_DIR="/backups/mysql"
DATE=$(date +%Y%m%d_%H%M%S)
RETENTION_DAYS=30

mkdir -p "$BACKUP_DIR"

# mysqldump로 전체 백업
docker exec mysql mysqldump \
  -u root \
  -p"$MYSQL_ROOT_PASSWORD" \
  --single-transaction \
  --routines \
  --triggers \
  mmp_prod > "$BACKUP_DIR/mmp_prod_$DATE.sql"

# 압축
gzip "$BACKUP_DIR/mmp_prod_$DATE.sql"

# 오래된 백업 삭제
find "$BACKUP_DIR" -name "*.sql.gz" -mtime +$RETENTION_DAYS -delete

echo "[$(date)] MySQL backup completed: mmp_prod_$DATE.sql.gz"
```

### 6-2. 백업 스케줄 (Crontab)

```bash
# 매일 새벽 2시 MySQL 백업
0 2 * * * /opt/scripts/backup-mysql.sh >> /var/log/backup.log 2>&1

# 매주 일요일 녹화 파일 목록 스냅샷
0 4 * * 0 find /recordings -name "*.mp4" -ls > /backups/recordings-manifest-$(date +\%Y\%m\%d).txt
```

### 6-3. 백업 검증

```bash
# 백업 파일 무결성 확인
gunzip -t /backups/mysql/mmp_prod_20260527_020000.sql.gz

# 테스트 복원 (스테이징 환경)
gunzip -c /backups/mysql/mmp_prod_20260527_020000.sql.gz | \
  docker exec -i mysql-staging mysql -u root -p"$PASSWORD" mmp_staging
```

## 7. 장애 대응 절차

### 7-1. 장애 유형별 대응

| 장애 유형 | 증상 | 대응 절차 |
|----------|------|----------|
| 컨테이너 중지 | 서비스 접근 불가 | `docker compose up -d [서비스명]` |
| DB 연결 실패 | API 500 에러 | MySQL 컨테이너 상태 확인 후 재시작 |
| 디스크 부족 | 녹화 중단, DB 쓰기 실패 | 오래된 녹화 파일 긴급 삭제 |
| 메모리 초과 | OOM Kill | 리소스 제한 상향, 메모리 누수 점검 |
| 카메라 연결 끊김 | 영상 표시 안 됨 | 카메라 네트워크/전원 확인 |

### 7-2. 컨테이너 재시작 절차

```bash
# 1. 현재 상태 확인
docker compose -f docker-compose.prod.yml ps

# 2. 특정 서비스 재시작
docker compose -f docker-compose.prod.yml restart mmp-server

# 3. 전체 재시작 (필요 시)
docker compose -f docker-compose.prod.yml down
docker compose -f docker-compose.prod.yml up -d

# 4. 로그 확인
docker logs --tail 50 mmp-server
```

### 7-3. 헬스체크 기반 자동 복구

Docker Compose의 `healthcheck` 설정으로 비정상 컨테이너를 자동 감지하고, `restart: always` 정책과 함께 자동 복구를 수행한다.

```bash
# 헬스 상태 확인
docker inspect --format='{{.State.Health.Status}}' mmp-server
# 결과: healthy | unhealthy | starting
```

### 8-2. Nginx SSL 설정

```nginx
server {
    listen 80;
    server_name vertiport.knu.ac.kr;
    return 301 https://$server_name$request_uri;
}

server {
    listen 443 ssl http2;
    server_name vertiport.knu.ac.kr;

    ssl_certificate     /etc/letsencrypt/live/vertiport.knu.ac.kr/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/vertiport.knu.ac.kr/privkey.pem;
    ssl_protocols       TLSv1.2 TLSv1.3;
    ssl_ciphers         HIGH:!aNULL:!MD5;

    location / {
        proxy_pass http://mmp-client:80;
    }

    location /api/ {
        proxy_pass http://mmp-server:3000;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
    }

    location /certbot/ {
        root /var/www;
    }
}
```

## 9. 보안 체크리스트

### 9-1. 인프라 보안

| 항목 | 상태 | 설명 |
|------|------|------|
| SSH 키 인증 | 필수 | 비밀번호 로그인 비활성화 |
| 방화벽 (ufw/iptables) | 필수 | 필요한 포트만 개방 (80, 443, 8554) |
| Docker 소켓 보호 | 필수 | 외부 노출 금지 |
| 비밀 정보 .env 분리 | 필수 | Git에 커밋하지 않음 |
| 정기 OS 업데이트 | 권장 | 보안 패치 적용 |

### 9-2. 애플리케이션 보안

| 항목 | 상태 | 설명 |
|------|------|------|
| JWT 토큰 인증 | 필수 | 모든 API에 인증 적용 |
| CORS 설정 | 필수 | 허용된 도메인만 접근 |
| Rate Limiting | 권장 | 로그인 시도 제한 (5회/분) |
| 입력값 검증 | 필수 | DTO validation (class-validator) |
| SQL Injection 방지 | 필수 | Prisma ORM 사용으로 자동 방어 |
| HTTPS 강제 | 필수 | HTTP -> HTTPS 리다이렉트 |

### 9-3. 카메라 네트워크 보안

```
┌─────────────────────────────────────────┐
│          운영 네트워크 (VLAN 10)          │
│  ┌──────────┐  ┌──────────┐             │
│  │ 운영 서버  │  │ 관제 PC  │             │
│  └─────┬────┘  └──────────┘             │
│        │                                 │
├────────┼─────────────────────────────────┤
│        │    카메라 네트워크 (VLAN 20)      │
│  ┌─────┴────┐                            │
│  │ 서버      │──── 카메라 전용 NIC         │
│  │(듀얼 NIC) │                            │
│  └──────────┘                            │
│     ┌──────┐ ┌──────┐ ┌──────┐          │
│     │Cam 1 │ │Cam 2 │ │Cam 3 │          │
│     └──────┘ └──────┘ └──────┘          │
└─────────────────────────────────────────┘
```

- 카메라 네트워크를 VLAN으로 분리하여 외부 접근을 차단한다.
- 서버는 듀얼 NIC으로 운영 네트워크와 카메라 네트워크에 각각 연결한다.
- 카메라 기본 비밀번호를 반드시 변경하고, 불필요한 서비스(FTP, Telnet)를 비활성화한다.

### 9-4. 포트 개방 정책

| 포트 | 프로토콜 | 용도 | 외부 노출 |
|------|---------|------|----------|
| 80 | TCP | HTTP (HTTPS 리다이렉트) | O |
| 443 | TCP | HTTPS | O |
| 3000 | TCP | NestJS API | X (Nginx 프록시) |
| 3306 | TCP | MySQL | X (내부 전용) |
| 8554 | TCP | RTSP (MediaMTX) | X (내부 전용) |
| 8888 | TCP | HLS (MediaMTX) | X (Nginx 프록시) |
| 8889 | TCP | WebRTC (MediaMTX) | 조건부 |
